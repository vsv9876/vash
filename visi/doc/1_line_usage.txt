char   *idxp[];         /* - textual representation of every index value; */

LINE line = { size, line, colu, flag, attr, cvts, cvtf, test, varl };
LINE page[] = { {line1}, {line2}, ..., { 0 } };

vmain() {
	LINE *cline;    /* pointer to current line (pointed by cursor) */
	int   cpos;     /* current position on editing start */

	w_line( &line);   r_line( &line, &cpos );
	w_page( page );   r_page( page, &cline, &cpos );
}
